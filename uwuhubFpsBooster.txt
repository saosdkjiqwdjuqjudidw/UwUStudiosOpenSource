local function fn2(arg)
	while arg and arg ~= Workspace and arg ~= game do
		local name = arg.Name
		if name == "MainPart" or name == "Walls" or name == "Baseplate" or name:find("Trash") or name:find("trash") or arg:IsA("SpawnLocation") then
			return true
		end
		arg = arg.Parent
	end

	return false
end

	end

	return false
end

local function fn4(arg)

end

local function fn5(descendant)
	if not flag then
		return false
	end
	local isDecal = descendant:IsA("Decal") or descendant:IsA("Texture") or descendant:IsA("SurfaceAppearance")
	if isDecal and fn2(descendant) then
		return false
	end

	if descendant:IsA("Sky") and descendant.Name == "B" then
		return false
	end

	if descendant:IsA("ParticleEmitter") or descendant:IsA("Trail") or descendant:IsA("Smoke") or descendant:IsA("Fire") or descendant:IsA("Sparkles") or descendant:IsA("Beam") or descendant:IsA("Light") or descendant:IsA("Highlight") or descendant:IsA("Explosion") or descendant:IsA("PostProcessEffect") or descendant:IsA("Atmosphere") or descendant:IsA("ColorCorrectionEffect") or descendant:IsA("BlurEffect") or descendant:IsA("BloomEffect") or descendant:IsA("SunRaysEffect") or descendant:IsA("Clouds") or isDecal then
		pcall(function()
			descendant.Enabled = false
			descendant.Parent = nil
			descendant:Destroy()
		end)

		return true
	end

	return false
end

local function fn6(arg)
	if arg:IsA("BasePart") and not fn2(arg) and not fn3(arg) then
		if tbl2[arg] == nil then
			tbl2[arg] = { cf = arg.CFrame, a = arg.Anchored, c = arg.CanCollide }

			pcall(function()
				arg.CanCollide = false
				arg.Anchored = true
				arg.CFrame = arg.CFrame + vector
			end)
		end
	end
end

local function fn7(descendant)
	if not flag then
		return
	end

	if fn5(descendant) then
		return
	end

	if descendant:IsA("BasePart") then
		if tbl[descendant] == nil then
			tbl[descendant] = {
				m = descendant.Material,
				cs = descendant.CastShadow,
				r = descendant.Reflectance,
				tex = descendant:IsA("MeshPart") and descendant.TextureID or nil,
			}
		end

		descendant.CastShadow = false
		descendant.Reflectance = 0

		if not fn2(descendant) then
			descendant.Material = Enum.Material.SmoothPlastic

			if descendant:IsA("MeshPart") then
				descendant.TextureID = ""
			end
		end

		if not fn3(descendant) and not fn2(descendant) and descendant.Name ~= "Baseplate" then
			local parent = descendant.Parent
			local flag3

			while true do
				local flag4 = parent and parent ~= Workspace and parent ~= game
				flag3 = false

				if flag4 then
					if tbl6[parent.Name] then
						flag3 = true
						break
					else
						parent = parent.Parent
						continue
					end
				end

				break
			end

			if flag3 then
				pcall(function()
					descendant:Destroy()
				end)
			elseif fn4(descendant) then
				fn6(descendant)
			end
		end
	end
end

		local v = descendants[i]

		if v:IsA("BasePart") then
			if not fn2(v) and not fn3(v) and fn4(v) then
				fn6(v)

					v.TextureID = ""

		end
